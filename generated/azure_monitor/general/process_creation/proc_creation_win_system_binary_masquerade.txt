# Windows System Binary Name Running From the Wrong Folder
# Sigma id d7ffbe3f-7838-53ca-89cc-d441a1088634, pipeline azure_monitor
SecurityEvent
| where ((NewProcessName endswith "\\smss.exe" or NewProcessName endswith "\\winlogon.exe" or NewProcessName endswith "\\dllhost.exe" or NewProcessName endswith "\\sihost.exe") and (not((NewProcessName startswith "C:\\Windows\\System32\\" or NewProcessName startswith "C:\\Windows\\SysWOW64\\")))) or (NewProcessName endswith "\\MicrosoftEdgeUpdate.exe" and (not((NewProcessName contains "\\Microsoft\\EdgeUpdate\\" or NewProcessName contains "\\Microsoft\\Temp\\"))))
