# hackingBuddyGPT Local Shell Command Markers Sent Through tmux
# Sigma id 9acfd2b0-21a2-5d83-958c-7901906c161e, pipeline ecs
from * metadata _id, _index, _version | where ends_with(process.executable, "/tmux") and process.command_line like "*send-keys*" and (process.command_line like "*echo 'CMDSTART*" or process.command_line like "*echo 'CMDEND*" or process.command_line like "*echo \"__CLEAR_*" or process.command_line like "*echo \"__END_*")
