# Web or Database Service Spawns a Shell or Download Tool on Windows
# Sigma id 6a64daf6-9e1b-57ab-b9fe-cf43e37123b8, pipeline ecs
from * metadata _id, _index, _version | where (ends_with(process.parent.executable.caseless, "\\w3wp.exe") or ends_with(process.parent.executable.caseless, "\\httpd.exe") or ends_with(process.parent.executable.caseless, "\\tomcat.exe") or ends_with(process.parent.executable.caseless, "\\php-cgi.exe") or ends_with(process.parent.executable.caseless, "\\sqlservr.exe") or ends_with(process.parent.executable.caseless, "\\mysqld.exe") or (ends_with(process.parent.executable.caseless, "\\java.exe") or ends_with(process.parent.executable.caseless, "\\node.exe")) and (user.name like "*IIS APPPOOL*" or user.name like "*NETWORK SERVICE*" or user.name like "*LOCAL SERVICE*")) and (ends_with(process.executable.caseless, "\\cmd.exe") or ends_with(process.executable.caseless, "\\powershell.exe") or ends_with(process.executable.caseless, "\\pwsh.exe") or ends_with(process.executable.caseless, "\\curl.exe") or ends_with(process.executable.caseless, "\\certutil.exe") or ends_with(process.executable.caseless, "\\bitsadmin.exe") or ends_with(process.executable.caseless, "\\whoami.exe") or ends_with(process.executable.caseless, "\\tasklist.exe"))
