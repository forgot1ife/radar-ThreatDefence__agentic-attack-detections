# CloudWatch Logs Searched for Secrets
# Sigma id 510a5a9c-a361-513d-b3f4-52af66cdf912, pipeline ecs
from * metadata _id, _index, _version | where event.provider=="logs.amazonaws.com" and event.action=="FilterLogEvents" and (aws.cloudtrail.request_parameters like "*password*" or aws.cloudtrail.request_parameters like "*AKIA*" or aws.cloudtrail.request_parameters like "*secret*" or aws.cloudtrail.request_parameters like "*token*")
