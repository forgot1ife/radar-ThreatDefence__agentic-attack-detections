# Windows System Binary Name Running From the Wrong Folder
# Sigma id d7ffbe3f-7838-53ca-89cc-d441a1088634, pipeline ecs
from * metadata _id, _index, _version | where (ends_with(process.executable.caseless, "\\smss.exe") or ends_with(process.executable.caseless, "\\winlogon.exe") or ends_with(process.executable.caseless, "\\dllhost.exe") or ends_with(process.executable.caseless, "\\sihost.exe")) and not (starts_with(process.executable.caseless, "C:\\Windows\\System32\\") or starts_with(process.executable.caseless, "C:\\Windows\\SysWOW64\\")) or ends_with(process.executable.caseless, "\\MicrosoftEdgeUpdate.exe") and not (process.executable.caseless like "*\\Microsoft\\EdgeUpdate\\*" or process.executable.caseless like "*\\Microsoft\\Temp\\*")
