# Burst of EC2 DryRun Calls From One Identity
# Sigma id 0786eda9-e147-504c-8520-874a7562ee4b, pipeline ecs
from * metadata _id, _index, _version | where event.provider=="ec2.amazonaws.com" and aws.cloudtrail.error_code=="Client.DryRunOperation"
| eval timebucket=date_trunc(5minutes, @timestamp) | stats event_count=count() by timebucket, aws.cloudtrail.user_identity.arn
| where event_count >= 10
