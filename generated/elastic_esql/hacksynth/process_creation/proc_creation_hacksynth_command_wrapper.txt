# HackSynth Agent Command Wrapper in a Container
# Sigma id 94fad7a0-f8b9-56de-af95-58b1ad11ef2e, pipeline ecs
from * metadata _id, _index, _version | where ends_with(process.executable, "/timeout") and process.command_line rlike "([^ ]*/)?timeout [0-9]+s /bin/bash -c .*" and (ends_with(process.parent.executable, "/containerd-shim") or ends_with(process.parent.executable, "/containerd-shim-runc-v2") or ends_with(process.parent.executable, "/runc"))
