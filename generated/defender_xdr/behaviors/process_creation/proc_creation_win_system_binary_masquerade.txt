# Windows System Binary Name Running From the Wrong Folder
# Sigma id d7ffbe3f-7838-53ca-89cc-d441a1088634, pipeline microsoft_xdr
DeviceProcessEvents
| where ((FolderPath endswith "\\smss.exe" or FolderPath endswith "\\winlogon.exe" or FolderPath endswith "\\dllhost.exe" or FolderPath endswith "\\sihost.exe") and (not((FolderPath startswith "C:\\Windows\\System32\\" or FolderPath startswith "C:\\Windows\\SysWOW64\\")))) or (FolderPath endswith "\\MicrosoftEdgeUpdate.exe" and (not((FolderPath contains "\\Microsoft\\EdgeUpdate\\" or FolderPath contains "\\Microsoft\\Temp\\"))))
